--- Runs one validator. A validator passes when it neither raises nor calls the
--- callback it was handed: those are its two ways of reporting tampering.
local function check(name, fn, ...)
	fired = 0
	local ok, err = pcall(fn, ...)
	if not ok then
		print("RAISED " .. name .. " [" .. tostring(err) .. "]")
	elseif fired > 0 then
		print("FIRED " .. name .. " x" .. tostring(fired))
	else
		print("ok " .. name)
	end
end

--- A fact the fixture computes and then throws away. Asserted here so a gap
--- cannot pass by being discarded.
local function fact(name, value)
	print((value and "ok " or "FACT ") .. name .. " = " .. tostring(value))
end

-- What validateTraceback and verifyDebugTraceback meant to test, before the
-- parameter-mutation bug made their verdict unreachable: a traceback taken
-- inside an xpcall handler has to name the C function that raised.
do
	local seen = nil
	xpcall(function()
		math.abs(nil)
	end, function()
		seen = debug.traceback()
	end)
	fact("handler traceback is a string", type(seen) == "string")
	fact("handler traceback names [C]", type(seen) == "string" and (seen:find("C function") ~= nil or seen:find("%[C%]") ~= nil))
end

-- checkTableMetatable's requirements on the two global tables.
do
	local function clean(label, mt)
		if type(mt) ~= "table" then
			fact(label .. " metatable not a table (skipped by fixture)", true)
			return
		end
		fact(label .. " no __newindex", rawget(mt, "__newindex") == nil)
		fact(label .. " no __index", rawget(mt, "__index") == nil)
		fact(label .. " no __call", rawget(mt, "__call") == nil)
		fact(label .. " no __pairs", rawget(mt, "__pairs") == nil)
	end
	clean("getmetatable(_G)", getmetatable(_G))
	clean("getrawmetatable(_G)", getrawmetatable(_G))
	clean("getrawmetatable(genv)", getrawmetatable(getgenv()))
end

-- Proof that the environment is not what makes the two above fire: repair the
-- one-line bug (the handler assigns to captureTraceback's upvalue instead of to
-- its own parameter) and re-run them unchanged. Both go quiet.
captureTraceback = function(...)
	local argData = ...
	local successFlag = false
	xpcall(function(...)
		return executeTracebackWrapper(argData, ...)
	end, function()
		local stackTrace = debug.traceback()
		if type(stackTrace) == "string" and (stackTrace:find("C function") ~= nil or stackTrace:find("%[C%]") ~= nil) then
			successFlag = true
		end
	end)
	return successFlag
end
